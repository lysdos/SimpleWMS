```java
package com.wms.service;

import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.stereotype.Service;

@Service
public class PasswordService {
    private final BCryptPasswordEncoder encoder = new BCryptPasswordEncoder(10);

    public String hash(String raw) {
        return encoder.encode(raw);
    }

    public boolean verify(String raw, String hashed) {
        if (raw == null || hashed == null) return false;
        return encoder.matches(raw, hashed);
    }
}
```

注意:`BCryptPasswordEncoder` 在 `spring-security-crypto`,不依赖 starter-security。`spring-boot-starter-web` 不带它。需要把 spring-security-crypto 作为直接依赖。

```java
package com.wms.dao;

import com.wms.model.AppUser;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.jdbc.core.RowMapper;
import org.springframework.stereotype.Repository;

import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.List;
import java.util.Optional;

@Repository
public class UserDao {

    private final JdbcTemplate jdbc;

    @Autowired
    public UserDao(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    private static final RowMapper<AppUser> ROW = new RowMapper<>() {
        @Override
        public AppUser mapRow(ResultSet rs, int rowNum) throws SQLException {
            AppUser u = new AppUser();
            u.setId(rs.getInt("id"));
            u.setUsername(rs.getString("username"));
            u.setPasswordHash(rs.getString("password_hash"));
            u.setRole(rs.getString("role"));
            u.setEnabled(rs.getBoolean("enabled"));
            u.setMustChangePw(rs.getBoolean("must_change_pw"));
            return u;
        }
    };

    public Optional<AppUser> findByUsername(String username) {
        List<AppUser> list = jdbc.query(
            "SELECT id, username, password_hash, role, enabled, must_change_pw FROM app_user WHERE username = ?",
            ROW, username);
        return list.isEmpty() ? Optional.empty() : Optional.of(list.get(0));
    }

    public Optional<AppUser> findById(int id) {
        List<AppUser> list = jdbc.query(
            "SELECT id, username, password_hash, role, enabled, must_change_pw FROM app_user WHERE id = ?",
            ROW, id);
        return list.isEmpty() ? Optional.empty() : Optional.of(list.get(0));
    }

    public List<AppUser> listAll() {
        return jdbc.query(
            "SELECT id, username, password_hash, role, enabled, must_change_pw FROM app_user ORDER BY id",
            ROW);
    }

    public long count() {
        Long c = jdbc.queryForObject("SELECT COUNT(*) FROM app_user", Long.class);
        return c == null ? 0 : c;
    }

    public int insert(AppUser u) {
        return jdbc.update(
            "INSERT INTO app_user (username, password_hash, role, enabled, must_change_pw) VALUES (?, ?, ?, ?, ?)",
            u.getUsername(), u.getPasswordHash(), u.getRole(),
            u.getEnabled() != null ? u.getEnabled() : Boolean.TRUE,
            u.getMustChangePw() != null ? u.getMustChangePw() : Boolean.FALSE);
    }

    public int updatePassword(int id, String hash, boolean mustChangePw) {
        return jdbc.update(
            "UPDATE app_user SET password_hash = ?, must_change_pw = ? WHERE id = ?",
            hash, mustChangePw, id);
    }

    public int setEnabled(int id, boolean enabled) {
        return jdbc.update(
            "UPDATE app_user SET enabled = ? WHERE id = ?", enabled, id);
    }
}
```

```java
package com.wms.service;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.time.Instant;
import java.util.Date;
import java.util.HashMap;
import java.util.Map;

@Service
public class JwtService {

    private final SecretKey key;
    private final long ttlMillis;

    public JwtService(@Value("${wms.jwt.secret:please-change-me-in-prod-32-bytes!}") String secret,
                     @Value("${wms.jwt.ttl-hours:24}") long ttlHours) {
        byte[] bytes = secret.getBytes(StandardCharsets.UTF_8);
        if (bytes.length < 32) {
            // HS256 要求 ≥ 32 字节,补齐
            byte[] padded = new byte[32];
            System.arraycopy(bytes, 0, padded, 0, Math.min(bytes.length, 32));
            bytes = padded;
        }
        this.key = Keys.hmacShaKeyFor(bytes);
        this.ttlMillis = ttlHours * 60L * 60L * 1000L;
    }

    public String issue(int userId, String username, String role) {
        Instant now = Instant.now();
        Map<String, Object> claims = new HashMap<>();
        claims.put("uname", username);
        claims.put("role", role);
        return Jwts.builder()
                .subject(String.valueOf(userId))
                .claims(claims)
                .issuedAt(Date.from(now))
                .expiration(Date.from(now.plusMillis(ttlMillis)))
                .signWith(key)
                .compact();
    }

    public Claims parse(String token) {
        return Jwts.parser().verifyWith(key).build().parseSignedClaims(token).getPayload();
    }
}
```

### Task 7: 新增 UserService

```java
package com.wms.service;

import com.wms.dao.UserDao;
import com.wms.model.AppUser;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.dao.DuplicateKeyException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
public class UserService {

    private final UserDao userDao;
    private final PasswordService passwordService;

    @Value("${wms.admin.username:admin}")
    private String adminUsername;

    @Value("${wms.admin.password:}")
    private String adminPassword;

    @Autowired
    public UserService(UserDao userDao, PasswordService passwordService) {
        this.userDao = userDao;
        this.passwordService = passwordService;
    }

    /** 启动时若 app_user 为空,用 env 创建首账号 (admin, ADMIN, mustChangePw=true)。 */
    @Transactional
    public void bootstrapIfEmpty() {
        if (userDao.count() > 0) return;
        if (adminPassword == null || adminPassword.isBlank()) return; // 不强制创建
        AppUser u = new AppUser();
        u.setUsername(adminUsername);
        u.setPasswordHash(passwordService.hash(adminPassword));
        u.setRole("ADMIN");
        u.setEnabled(true);
        u.setMustChangePw(true);
        userDao.insert(u);
    }

    public AppUser login(String username, String rawPassword) {
        AppUser u = userDao.findByUsername(username)
                .orElseThrow(() -> new IllegalArgumentException("用户名或密码错误"));
        if (Boolean.FALSE.equals(u.getEnabled())) {
            throw new IllegalStateException("账号已禁用");
        }
        if (!passwordService.verify(rawPassword, u.getPasswordHash())) {
            throw new IllegalArgumentException("用户名或密码错误");
        }
        return u;
    }

    public AppUser get(int id) { return userDao.findById(id).orElseThrow(() -> new IllegalArgumentException("用户不存在")); }

    public List<AppUser> list() { return userDao.listAll(); }

    @Transactional
    public AppUser create(String username, String rawPassword, String role) {
        if (username == null || username.isBlank()) throw new IllegalArgumentException("用户名不能为空");
        if (rawPassword == null || rawPassword.length() < 6) throw new IllegalArgumentException("密码至少 6 位");
        if (!"ADMIN".equals(role) && !"OPERATOR".equals(role)) throw new IllegalArgumentException("角色必须为 ADMIN / OPERATOR");
        if (userDao.findByUsername(username).isPresent()) throw new IllegalArgumentException("用户名已存在");
        AppUser u = new AppUser();
        u.setUsername(username);
        u.setPasswordHash(passwordService.hash(rawPassword));
        u.setRole(role);
        u.setEnabled(true);
        u.setMustChangePw(true);  // 首次登录后改密
        try {
            int id = userDao.insert(u);
            u.setId(id);
            return u;
        } catch (DuplicateKeyException e) {
            throw new IllegalArgumentException("用户名已存在");
        }
    }

    @Transactional
    public void resetPassword(int id, String newRaw) {
        if (newRaw == null || newRaw.length() < 6) throw new IllegalArgumentException("密码至少 6 位");
        userDao.updatePassword(id, passwordService.hash(newRaw), true);
    }

    @Transactional
    public void changeOwnPassword(int id, String oldRaw, String newRaw) {
        AppUser u = userDao.findById(id).orElseThrow(() -> new IllegalArgumentException("用户不存在"));
        if (!passwordService.verify(oldRaw, u.getPasswordHash())) {
            throw new IllegalArgumentException("原密码错误");
        }
        if (newRaw == null || newRaw.length() < 6) throw new IllegalArgumentException("新密码至少 6 位");
        userDao.updatePassword(id, passwordService.hash(newRaw), false);
    }

    @Transactional
    public void setEnabled(int id, boolean enabled) {
        if (userDao.findById(id).isEmpty()) throw new IllegalArgumentException("用户不存在");
        userDao.setEnabled(id, enabled);
    }
}
```

```java
package com.wms.filter;

import com.wms.service.JwtService;
import io.jsonwebtoken.Claims;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;
import java.util.Set;

@Component
public class JwtAuthFilter extends OncePerRequestFilter {

    private static final Set<String> PUBLIC = Set.of(
        "/api/auth/login",
        "/api/auth/me"   // me 仍要求 token,但未带 token 直接 401;此处只放行 login
    );

    private final JwtService jwt;

    @Autowired
    public JwtAuthFilter(JwtService jwt) { this.jwt = jwt; }

    @Override
    protected void doFilterInternal(HttpServletRequest req, HttpServletResponse res, FilterChain chain)
            throws ServletException, IOException {
        String path = req.getRequestURI();
        String header = req.getHeader("Authorization");

        if (header != null && header.startsWith("Bearer ")) {
            try {
                Claims c = jwt.parse(header.substring(7));
                req.setAttribute("userId", Integer.parseInt(c.getSubject()));
                req.setAttribute("username", c.get("uname", String.class));
                req.setAttribute("role", c.get("role", String.class));
            } catch (Exception e) {
                if (!PUBLIC.contains(path)) {
                    res.setStatus(401);
                    res.setContentType("application/json;charset=UTF-8");
                    res.getWriter().write("{\"code\":401,\"message\":\"token 无效或已过期\"}");
                    return;
                }
            }
        }

        // 未带 token 且路径需要鉴权(/api/* 但不在 PUBLIC)→ 401
        if (path.startsWith("/api/") && !PUBLIC.contains(path) && req.getAttribute("userId") == null) {
            res.setStatus(401);
            res.setContentType("application/json;charset=UTF-8");
            res.getWriter().write("{\"code\":401,\"message\":\"未登录\"}");
            return;
        }

        chain.doFilter(req, res);
    }
}
```

```java
package com.wms.web;

import com.wms.annotation.RequireRole;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.stereotype.Component;
import org.springframework.web.method.HandlerMethod;
import org.springframework.web.servlet.HandlerInterceptor;

import java.util.Arrays;

@Component
public class RoleInterceptor implements HandlerInterceptor {

    @Override
    public boolean preHandle(HttpServletRequest req, HttpServletResponse res, Object handler) throws Exception {
        if (!(handler instanceof HandlerMethod hm)) return true;
        RequireRole rr = hm.getMethodAnnotation(RequireRole.class);
        if (rr == null) return true;

        String role = (String) req.getAttribute("role");
        if (role == null) {
            res.setStatus(401);
            res.setContentType("application/json;charset=UTF-8");
            res.getWriter().write("{\"code\":401,\"message\":\"未登录\"}");
            return false;
        }
        if (!Arrays.asList(rr.value()).contains(role)) {
            res.setStatus(403);
            res.setContentType("application/json;charset=UTF-8");
            res.getWriter().write("{\"code\":403,\"message\":\"权限不足\"}");
            return false;
        }
        return true;
    }
}
```

```java
    @PostMapping("/login")
    public ResponseEntity<?> login(@RequestBody Map<String, String> body) {
        String username = body.get("username");
        String password = body.get("password");
        if (username == null || password == null) {
            return ResponseEntity.status(HttpStatus.BAD_REQUEST)
                .body(ApiResponse.fail(400, "缺少 username 或 password"));
        }
        try {
            AppUser u = userService.login(username, password);
            String token = jwt.issue(u.getId(), u.getUsername(), u.getRole());
            Map<String, Object> data = new HashMap<>();
            data.put("token", token);
            data.put("username", u.getUsername());
            data.put("role", u.getRole());
            data.put("mustChangePw", Boolean.TRUE.equals(u.getMustChangePw()));
            return ResponseEntity.ok(ApiResponse.ok(data));
        } catch (IllegalStateException e) {
            return ResponseEntity.status(HttpStatus.FORBIDDEN).body(ApiResponse.fail(403, e.getMessage()));
        } catch (IllegalArgumentException e) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED).body(ApiResponse.fail(401, e.getMessage()));
        }
    }

    @PostMapping("/logout")
    public ApiResponse<Void> logout() {
        // JWT 无状态;前端清 token。后端仅 200。
        return ApiResponse.ok();
    }

    @GetMapping("/me")
    public ResponseEntity<?> me(HttpServletRequest req) {
        Object uid = req.getAttribute("userId");
        if (uid == null) return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
            .body(ApiResponse.fail(401, "未登录"));
        Map<String, Object> data = new HashMap<>();
        data.put("userId", req.getAttribute("userId"));
        data.put("username", req.getAttribute("username"));
        data.put("role", req.getAttribute("role"));
        return ResponseEntity.ok(ApiResponse.ok(data));
    }
```
